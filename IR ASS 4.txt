# Import required libraries
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix


# 1. Create dataset containing Spam and Ham messages
dataset = [
    ("Win a $1000 Walmart gift card! Click here now to claim your cash reward prize.", "spam"),
    ("Urgent: Your account security has been compromised. Verify your password now.", "spam"),
    ("Congratulations! You have been selected for a free lottery ticket worth 50,000.", "spam"),
    ("Exclusive loan offer approved with 0% interest! Text YES to claim your funds.", "spam"),
    ("Claim your free entry to win a brand new car today. Call this number now.", "spam"),
    ("Free trial offer expiring soon! Click the link below to renew your access.", "spam"),
    ("You have won a guaranteed cash prize! Reply WIN to 88202 to collect.", "spam"),
    ("Double your income working from home. No experience needed, apply fast.", "spam"),
    ("Hey, are we still meeting for lunch at 1 PM today?", "ham"),
    ("Can you send me the updated project report and presentation deck?", "ham"),
    ("The team meeting has been rescheduled to tomorrow at 10 AM.", "ham"),
    ("Please review the attached invoice and let me know if any changes are needed.", "ham"),
    ("Let's catch up over coffee this weekend if you are free.", "ham"),
    ("Don't forget to submit the assignment before midnight today.", "ham"),
    ("Thanks for your help with the machine learning assignment yesterday!", "ham"),
    ("Hi, I just pushed the latest commit to the repository.", "ham")
] * 20


# Convert dataset into a Pandas DataFrame
df = pd.DataFrame(dataset, columns=["text", "label"])


# Convert categorical labels into numerical values
# Ham = 0 and Spam = 1
df["label_num"] = df["label"].map({"ham": 0, "spam": 1})


# 2. Split dataset into training and testing data
X_train, X_test, y_train, y_test = train_test_split(
    df["text"], df["label_num"],
    test_size=0.25,
    random_state=42,
    stratify=df["label_num"]
)


# 3. Convert text into numerical features using TF-IDF
tfidf = TfidfVectorizer(stop_words="english", lowercase=True)

# Fit vectorizer on training data and transform it
X_train_vec = tfidf.fit_transform(X_train)

# Transform testing data using the same vectorizer
X_test_vec = tfidf.transform(X_test)


# 4. Create and train Multinomial Naive Bayes classifier
nb_model = MultinomialNB()

nb_model.fit(X_train_vec, y_train)


# 5. Predict labels for test data
y_pred = nb_model.predict(X_test_vec)


# Calculate model accuracy
acc = accuracy_score(y_test, y_pred)

print("=== ASSIGNMENT 4: SPAM FILTER CLASSIFIER ===")
print(f"Overall Accuracy: {acc * 100:.2f}%\n")


# Display classification performance
print("Classification Report:")

print(classification_report(
    y_test, y_pred,
    target_names=["Ham (Legitimate)", "Spam"]
))


# 6. Test the model using new unseen email messages
test_emails = [
    "Win a free lottery prize worth 1000 dollars now!",
    "Hey, can we schedule the review meeting for tomorrow morning?"
]


# Convert new emails into TF-IDF numerical features
sample_vec = tfidf.transform(test_emails)


# Predict whether emails are Spam or Ham
predictions = nb_model.predict(sample_vec)


print("\n--- Predictions on New Incoming Emails ---")


# Display final predictions
for text, pred in zip(test_emails, predictions):

    # Convert numerical prediction into category
    tag = "SPAM" if pred == 1 else "HAM"

    print(f'Text: "{text}" -> Classified as: [{tag}]')

## A. Basic Bayesian Network Questions

**Q1. What is a Bayesian Network?**

Ans: A Bayesian Network is a probabilistic graphical model that represents relationships between different variables using a directed acyclic graph.

**Q2. What is the full form of DAG?**

Ans: Directed Acyclic Graph.

**Q3. What is a DAG?**

Ans: A DAG is a graph in which edges have directions and there are no cycles or circular paths.

**Q4. Why do we use a Bayesian Network?**

Ans: We use Bayesian Networks to represent uncertain relationships and calculate probabilities based on available evidence.

**Q5. What is conditional probability?**

Ans: Conditional probability is the probability of an event occurring when another event is already known.

**Q6. What is Bayes' Theorem?**

Ans: Bayes' Theorem is used to calculate the probability of an event based on previously known information.

Formula:

P(A|B) = P(B|A) × P(A) / P(B)

**Q7. What is the objective of this assignment?**

Ans: To construct a Bayesian Network using heart disease medical data and calculate the probability of heart disease based on medical evidence.

**Q8. What is a node in a Bayesian Network?**

Ans: A node represents a variable in the network, such as age, cholesterol or heart disease.

**Q9. What is an edge in a Bayesian Network?**

Ans: An edge represents a directed relationship between two variables.

**Q10. What is a Conditional Probability Table (CPT)?**

Ans: A CPT is a table that stores the conditional probabilities of a variable based on its parent variables.

## C. Important Conceptual Questions

**Q20. What is Maximum Likelihood Estimation (MLE)?**

Ans: MLE is a statistical method used to estimate model parameters by finding the values that make the observed data most likely.

**Q21. What is the difference between prior and posterior probability?**

Ans:
- Prior probability: Probability before considering new evidence.
- Posterior probability: Updated probability after considering new evidence.

**Q22. What is the difference between Bayesian Network and Naive Bayes?**

Ans:
- Bayesian Network represents relationships between variables using a graphical structure.
- Naive Bayes assumes that features are conditionally independent given the class.

**Q23. What is probabilistic inference?**

Ans: It is the process of calculating unknown probabilities using known information and relationships between variables.

**Q24. What is Variable Elimination used for?**

Ans: It is used to calculate the required probability by eliminating unnecessary variables from the probability calculations.

**Q25. What is the difference between training and inference?**

Ans:
- Training: Learning probability values from the dataset.
- Inference: Calculating probabilities using the trained model.

**Q26. Can Bayesian Networks handle uncertainty?**

Ans: Yes. Bayesian Networks are specifically designed to represent and reason under uncertainty.

**Q27. What happens if evidence is not provided?**

Ans: The model calculates the probability distribution without any additional evidence, using the learned probability values.

**Q28. What are the advantages of Bayesian Networks?**

Ans:
1. Handle uncertain information.
2. Represent relationships between variables.
3. Support probabilistic reasoning.
4. Can work with incomplete evidence.

**Q29. What are the applications of Bayesian Networks?**

Ans:
1. Medical diagnosis.
2. Weather prediction.
3. Risk analysis.
4. Fault detection.
5. Decision support systems.
