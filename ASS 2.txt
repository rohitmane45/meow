import requests
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt


def fetch_weather_forecast(city="Pune", api_key="YOUR_API_KEY"):

    url = f"https://api.openweathermap.org/data/2.5/forecast?q={city}&appid={api_key}&units=metric"

    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        data = response.json()

        records = []

        for entry in data["list"]:
            records.append({
                "timestamp": pd.to_datetime(entry["dt_txt"]),
                "temperature_c": entry["main"]["temp"],
                "humidity_pct": entry["main"]["humidity"],
                "pressure_hpa": entry["main"]["pressure"],
                "wind_speed_mps": entry["wind"]["speed"]
            })

        return pd.DataFrame(records)

    except Exception as e:

        print(f"Notice: Simulated data used ({e}).")

        dates = pd.date_range(
            start="2026-10-01 00:00",
            periods=24,
            freq="3h"
        )

        return pd.DataFrame({
            "timestamp": dates,
            "temperature_c": np.round(
                24.0 + 6.0 * np.sin(np.linspace(0, 3*np.pi, 24)), 2
            ),
            "humidity_pct": np.round(
                65.0 - 15.0 * np.sin(np.linspace(0, 3*np.pi, 24)), 1
            ),
            "pressure_hpa": np.round(
                1012.0 + np.random.normal(0, 1.2, 24), 1
            ),
            "wind_speed_mps": np.round(
                np.random.uniform(2.0, 7.5, 24), 2
            )
        })


df_weather = fetch_weather_forecast(

2. Python Code – With Comments and Explanation
# Import required libraries

import requests
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt


# Function to fetch weather forecast from OpenWeatherMap API

def fetch_weather_forecast(city="Pune", api_key="YOUR_API_KEY"):

    # API URL with city name and API key
    url = f"https://api.openweathermap.org/data/2.5/forecast?q={city}&appid={api_key}&units=metric"

    try:

        # Send GET request to API
        response = requests.get(url, timeout=10)

        # Check whether request was successful
        response.raise_for_status()

        # Convert API response into JSON
        data = response.json()

        records = []

        # Extract required weather attributes
        for entry in data["list"]:

            records.append({
                "timestamp": pd.to_datetime(entry["dt_txt"]),
                "temperature_c": entry["main"]["temp"],
                "humidity_pct": entry["main"]["humidity"],
                "pressure_hpa": entry["main"]["pressure"],
                "wind_speed_mps": entry["wind"]["speed"]
            })

        # Convert records into DataFrame
        return pd.DataFrame(records)

    except Exception as e:

        # Generate simulated data if API is unavailable
        print(f"Notice: Simulated data used ({e}).")

        dates = pd.date_range(
            start="2026-10-01 00:00",
            periods=24,
            freq="3h"
        )

        # Create sample weather dataset
        return pd.DataFrame({
            "timestamp": dates,

            "temperature_c": np.round(
                24.0 + 6.0 * np.sin(np.linspace(0, 3*np.pi, 24)), 2
            ),

            "humidity_pct": np.round(
                65.0 - 15.0 * np.sin(np.linspace(0, 3*np.pi, 24)), 1
            ),

            "pressure_hpa": np.round(
                1012.0 + np.random.normal(0, 1.2, 24), 1
            ),

            "wind_speed_mps": np.round(
                np.random.uniform(2.0, 7.5, 24), 2
            )
        })


# Retrieve and process weather data

df_weather = fetch_weather_forecast(
    city="Pune",
    api_key="demo_key"
)


# Statistical Modeling and Summary

print("=== WEATHER ATTRIBUTES DESCRIPTIVE SUMMARY ===")

# Calculate count, mean, standard deviation, minimum and maximum
print(
    df_weather[

# Calculate correlation between weather attributes

print("\n--- Attribute Correlation Matrix ---")

)

plt.figure(figsize=(10, 5))

plt.ylabel("Value")

plt.grid(True, linestyle="--", alpha=0.5)

4. SPPU Practical Viva Questions and Answers
These questions are based on the oral questions in your manual (page 17) and the actual Python concepts used in Assignment 2.
A. Basic API Questions
Q1. What is OpenWeatherMap?
Answer: OpenWeatherMap is an online platform that provides weather information through APIs, such as temperature, humidity, pressure and wind speed.
Q2. What is a Web API?
Answer: A Web API is an interface that allows two different software applications to communicate and exchange data over the internet.
Q3. How do we interact with Web APIs?
Answer: We interact with Web APIs by sending HTTP requests, receiving responses from the server and processing the returned data.
Q4. What is an API key?
Answer: An API key is a unique identification code used to authenticate requests made to an API.
Q5. What is HTTP?
Answer: HTTP stands for Hypertext Transfer Protocol. It is a communication protocol used to transfer data between clients and servers.
B. Python and API Questions
Q6. Which Python library is used to call APIs in this program?
Answer: We use the requests library to send HTTP requests and receive responses from APIs.
Q7. What is a GET request?
Answer: A GET request is used to retrieve information from a server.
Q8. What is JSON?
Answer: JSON stands for JavaScript Object Notation. It is a lightweight data format used to exchange information between applications.
Q9. What is the use of response.json()?
Answer: It converts the API response into Python dictionary and list structures.
Q10. What is the use of requests.get()?
Answer: It sends a GET request to the specified API URL and retrieves the response.
Q11. What is the use of timeout=10?
Answer: It specifies that the request should wait up to 10 seconds before timing out.
Q12. Why do we use try-except?
Answer: It handles runtime errors and prevents the program from terminating unexpectedly.
Q13. What is an API endpoint?
Answer: An API endpoint is a specific URL through which an application accesses a particular API service.
C. Data Analysis Questions
Q14. What is Pandas?
Answer: Pandas is a Python library used for data manipulation, cleaning and analysis.
Q15. What is NumPy?
Answer: NumPy is a Python library used for numerical calculations and working with arrays.
Q16. What is Matplotlib?
Answer: Matplotlib is a Python library used to create graphs and visualizations.
Q17. What is descriptive statistics?
Answer: Descriptive statistics is the process of summarizing data using measures such as mean, minimum, maximum and standard deviation.
Q18. What is the use of describe()?
Answer: It provides statistical information about numerical columns, including count, mean, standard deviation, minimum, maximum and quartiles.
Q19. What is correlation?
Answer: Correlation measures the relationship between two numerical variables.
Q20. What is the use of .corr()?
Answer: It calculates the correlation between numerical columns in a DataFrame.
Q21. What is a time-series dataset?
Answer: A time-series dataset contains data collected or recorded over different time intervals.
Q22. Why do we use pd.to_datetime()?
Answer: It converts date and time values into a proper datetime format for time-based analysis.
D. Visualization Questions
Q23. Why do we use a line chart?
Answer: A line chart is used to show changes and trends in data over time.
Q24. What is the use of plt.figure()?
Answer: It creates a new figure for displaying a graph.
Q25. What is the use of plt.xlabel() and plt.ylabel()?
Answer: They are used to assign labels to the X-axis and Y-axis.
Q26. What is the use of plt.legend()?
Answer: It displays the labels of different plotted lines in a graph.
Q27. What is the use of plt.savefig()?
Answer: It saves the generated graph into an image file.
E. Important Conceptual Questions
Q28. What is humidity?
Answer: Humidity is the amount of water vapour present in the air. It is generally measured as a percentage.
Q29. What is atmospheric pressure?
Answer: Atmospheric pressure is the force exerted by the weight of air on the Earth's surface. It is commonly measured in hPa.
Q30. What is wind speed?
Answer: Wind speed is the rate at which air moves. In this program, it is measured in metres per second.
Q31. What is the difference between positive and negative correlation?
Answer:
•	Positive correlation: Both variables tend to increase or decrease together.
•	Negative correlation: When one variable increases, the other tends to decrease.
Q32. What does a correlation value of -0.97 indicate?
Answer: It indicates a very strong negative relationship between two variables.
Q33. Why is simulated data used in this program?
Answer: Simulated data is used as a fallback when the API request fails or the internet connection is unavailable. It allows us to test the program without depending on the API.
Q34. What is the difference between real-time data and forecast data?
Answer:
•	Real-time data represents current weather conditions.
•	Forecast data represents predicted weather conditions for future time intervals.
