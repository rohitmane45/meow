import numpy as np
from qiskit import QuantumCircuit, QuantumRegister, ClassicalRegister, transpile
from qiskit_aer import AerSimulator

def quantum_teleportation():

    qr = QuantumRegister(3, name="q")
    crz = ClassicalRegister(1, name="crz")
    crx = ClassicalRegister(1, name="crx")
    cr_target = ClassicalRegister(1, name="result")

    qc = QuantumCircuit(qr, crz, crx, cr_target)

    theta = np.pi / 3
    qc.ry(theta, qr[0])
    qc.barrier()

    qc.h(qr[1])
    qc.cx(qr[1], qr[2])
    qc.barrier()

    qc.cx(qr[0], qr[1])
    qc.h(qr[0])
    qc.barrier()

    qc.measure(qr[0], crz)
    qc.measure(qr[1], crx)
    qc.barrier()

    with qc.if_test((crx, 1)):
        qc.x(qr[2])

    with qc.if_test((crz, 1)):
        qc.z(qr[2])

    qc.barrier()

    qc.measure(qr[2], cr_target)

    return qc


qc = quantum_teleportation()

print("Circuit diagram:")
print(qc.draw(output="text"))

simulator = AerSimulator()
compiled_circuit = transpile(qc, simulator)

job = simulator.run(compiled_circuit, shots=10000)
result = job.result()
counts = result.get_counts()

bob_counts = {"0": 0, "1": 0}

for bitstring, count in counts.items():
    target_bit = bitstring.split()[0]
    bob_counts[target_bit] += count

print("\n--- Measurement Results for Bob's Qubit (q2) ---")
print("Total Shots: 10,000")
print(f"State |0>: {bob_counts['0']} counts ({bob_counts['0']/100:.2f}%)")
print(f"State |1>: {bob_counts['1']} counts ({bob_counts['1']/100:.2f}%)")
print("Theoretical: P(|0>) = 75.00%, P(|1>) = 25.00%")
Compatibility note: Your PDF uses the older .c_if() conditional-gate syntax. The code above uses Qiskit's newer if_test() syntax to support modern versions. The teleportation logic remains the same.

# Import required libraries
import numpy as np
from qiskit import QuantumCircuit, QuantumRegister, ClassicalRegister, transpile
from qiskit_aer import AerSimulator

# Define the quantum teleportation function
def quantum_teleportation():

    # Create 3 quantum bits
    qr = QuantumRegister(3, name="q")

    # Create classical registers for Alice's measurements
    crz = ClassicalRegister(1, name="crz")
    crx = ClassicalRegister(1, name="crx")

    # Create a classical register for Bob's final result
    cr_target = ClassicalRegister(1, name="result")

    # Create the quantum circuit
    qc = QuantumCircuit(qr, crz, crx, cr_target)

    # Prepare the initial quantum state on q0
    theta = np.pi / 3
    qc.ry(theta, qr[0])

    # Add barrier for circuit separation
    qc.barrier()

    # Create entanglement between Alice's q1 and Bob's q2
    qc.h(qr[1])
    qc.cx(qr[1], qr[2])

    qc.barrier()

    # Alice performs Bell-state measurement
    qc.cx(qr[0], qr[1])
    qc.h(qr[0])

    qc.barrier()

    # Measure Alice's two qubits
    qc.measure(qr[0], crz)
    qc.measure(qr[1], crx)

    qc.barrier()

    # Bob applies X correction if crx is 1
    with qc.if_test((crx, 1)):
        qc.x(qr[2])

    # Bob applies Z correction if crz is 1
    with qc.if_test((crz, 1)):
        qc.z(qr[2])

    qc.barrier()

    # Measure Bob's final qubit
    qc.measure(qr[2], cr_target)

    # Return the completed circuit
    return qc


# Generate the quantum teleportation circuit
qc = quantum_teleportation()

# Display the circuit diagram
print("Circuit diagram:")
print(qc.draw(output="text"))

# Create quantum simulator
simulator = AerSimulator()

# Compile the circuit
compiled_circuit = transpile(qc, simulator)

# Execute the circuit for 10000 shots
job = simulator.run(compiled_circuit, shots=10000)

# Get execution results
result = job.result()
counts = result.get_counts()

# Initialize Bob's measurement counts
bob_counts = {"0": 0, "1": 0}

# Extract Bob's final measurement results
for bitstring, count in counts.items():

    # Extract the first classical register
    target_bit = bitstring.split()[0]

    # Add the measurement count
    bob_counts[target_bit] += count

# Display final results
print("\n--- Measurement Results for Bob's Qubit (q2) ---")
print("Total Shots: 10,000")

print(f"State |0>: {bob_counts['0']} counts ({bob_counts['0']/100:.2f}%)")
print(f"State |1>: {bob_counts['1']} counts ({bob_counts['1']/100:.2f}%)")

print("Theoretical: P(|0>) = 75.00%, P(|1>) = 25.00%")

PART 3 — SPPU PRACTICAL VIVA QUESTIONS AND ANSWERS
Questions from the lab manual (page 16) are included, followed by additional important practical viva questions.
A. Basic Quantum Teleportation Questions
Q1. What is Quantum Teleportation?
Ans: Quantum teleportation is a process of transferring an unknown quantum state from one qubit to another using entanglement and classical communication.
Q2. What is the objective of this assignment?
Ans: To implement the quantum teleportation algorithm using Python and Qiskit.
Q3. What is Entanglement?
Ans: Entanglement is a quantum phenomenon in which two or more qubits become connected, so their measurement outcomes are correlated even when they are separated.
Q4. What is the role of Alice and Bob?
Ans:
•	Alice: Sender who has the original quantum state.
•	Bob: Receiver who reconstructs the quantum state on his qubit.
Q5. Why do we use quantum teleportation?
Ans: It is used to transfer quantum information from one location to another without physically sending the original qubit.
Q6. Does quantum teleportation transfer physical matter?
Ans: No. It transfers quantum information, not physical matter.
Q7. Can we copy a quantum state?
Ans: No. The no-cloning theorem states that an arbitrary unknown quantum state cannot be perfectly copied.
Q8. What is a Bell pair?
Ans: A Bell pair is a pair of maximally entangled qubits.
B. Quantum Gates Questions
Q9. Which gates are used in quantum teleportation?
Ans: The main gates used are:
1.	Hadamard (H) gate.
2.	CNOT (CX) gate.
3.	Pauli-X gate.
4.	Pauli-Z gate.
5.	Ry gate for preparing the initial state in this program.
Q10. What is a Hadamard gate?
Ans: It creates superposition by converting a computational basis state into an equal combination of 0 and 1.
Q11. What is a CNOT gate?
Ans: CNOT stands for Controlled NOT. It flips the target qubit when the control qubit is 1.
Q12. What is an X gate?
Ans: The X gate is a bit-flip gate. It converts 0 into 1 and 1 into 0.
Q13. What is a Z gate?
Ans: The Z gate is a phase-flip gate. It changes the phase of the |1⟩ state while keeping the measurement probabilities unchanged.
Q14. What is an Ry gate?
Ans: The Ry gate rotates a qubit around the Y-axis of the Bloch sphere.
Q15. What is a Barrier in Qiskit?
Ans: A barrier separates different sections of a quantum circuit and helps organize the circuit diagram.
C. Program-Related Questions
Q16. How many qubits are used in this program?
Ans: Three qubits are used.
Q17. What is the purpose of q0?
Ans: q0 contains the original quantum state that Alice wants to teleport.
Q18. What is the purpose of q1?
Ans: q1 is Alice's half of the entangled Bell pair.
Q19. What is the purpose of q2?
Ans: q2 is Bob's qubit, on which the teleported state is reconstructed.
Q20. Why do we use classical registers?
Ans: Classical registers store the measurement results obtained from Alice's qubits and Bob's final qubit.
Q21. Why do we use two classical bits for Alice?
Ans: Alice measures two qubits, so she needs two classical bits to store the results.
Q22. What is the purpose of qc.ry()?
Ans: It applies a rotation around the Y-axis to prepare the initial quantum state.
Q23. What does np.pi mean?
Ans: It represents the mathematical constant π using NumPy.
Q24. Why do we use shots=10000?
Ans: It executes the quantum circuit 10,000 times to obtain a statistical distribution of measurement outcomes.
Q25. What is AerSimulator?
Ans: AerSimulator is a local simulator used to execute quantum circuits without requiring a physical quantum computer.
Q26. What is transpile()?
Ans: It converts a quantum circuit into instructions supported by the selected backend.
D. Conceptual and Important Viva Questions
Q27. What is the no-cloning theorem?
Ans: It states that an unknown arbitrary quantum state cannot be copied perfectly.
Q28. What happens when a qubit is measured?
Ans: Its superposition collapses into one of the possible classical measurement outcomes.
Q29. Why does Bob need correction gates?
Ans: Because Alice's measurement changes the conditional state of Bob's qubit. Correction gates help Bob recover the original quantum state.
Q30. What are the possible measurement combinations of Alice's two qubits?
Ans: Four possible combinations:
•	00
•	01
•	10
•	11
Q31. What is classical communication in quantum teleportation?
Ans: Alice sends her two classical measurement bits to Bob so that he can apply the correct quantum operations.
Q32. Does quantum teleportation require entanglement?
Ans: Yes. Shared entanglement between Alice and Bob is essential for quantum teleportation.
Q33. Is quantum teleportation faster than light?
Ans: No. It requires classical communication, which cannot travel faster than light.
Q34. What is the theoretical probability of getting 0 and 1 in this program?
Ans:
•	Probability of 0 = 75%.
•	Probability of 1 = 25%.
Q35. Why is the output not exactly 75% and 25%?
Ans: Because measurement is probabilistic. With a finite number of shots, actual results can differ slightly from theoretical probabilities.
Q36. What is the difference between quantum teleportation and classical data transfer?
Ans: Classical data can be copied and transmitted directly, whereas quantum teleportation transfers an unknown quantum state using entanglement and classical communication.
