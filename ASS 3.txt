import numpy as np
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator

def build_qft_circuit(n_qubits=5):
    qc = QuantumCircuit(n_qubits, name=f"QFT_{n_qubits}")

    for target in range(n_qubits):
        qc.h(target)
        for control in range(target + 1, n_qubits):
            k = control - target + 1
            angle = 2 * np.pi / (2 ** k)
            qc.cp(angle, control, target)
        qc.barrier()

    for i in range(n_qubits // 2):
        qc.swap(i, n_qubits - i - 1)

    return qc


n = 5
qft_5 = build_qft_circuit(n)
iqft_5 = qft_5.inverse()
iqft_5.name = f"IQFT_{n}"

full_qc = QuantumCircuit(n, n)

input_val = 5
binary_repr = format(input_val, f'0{n}b')[::-1]

for qubit_idx, bit in enumerate(binary_repr):
    if bit == '1':
        full_qc.x(qubit_idx)

full_qc.barrier(label="Input State |5>")

full_qc.append(qft_5, range(n))
full_qc.barrier(label="QFT")

full_qc.append(iqft_5, range(n))
full_qc.barrier(label="IQFT")

full_qc.measure(range(n), range(n))

print("--- 5-Qubit QFT Subcircuit ---")
print(qft_5.draw(output="text"))

print("\n--- Full Test Circuit ---")
print(full_qc.draw(output="text"))

simulator = AerSimulator()
compiled_circuit = transpile(full_qc, simulator)
result = simulator.run(compiled_circuit, shots=1024).result()
counts = result.get_counts()

print("\n--- Measurement Output (1024 shots) ---")
for bitstring, count in counts.items():
    print(f"Bitstring: {bitstring} | Decimal: {int(bitstring, 2)} | Counts: {count}/1024")

# Import required libraries
import numpy as np
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator

# Function to build a Quantum Fourier Transform circuit
def build_qft_circuit(n_qubits=5):

    # Create a quantum circuit with the specified number of qubits
    qc = QuantumCircuit(n_qubits, name=f"QFT_{n_qubits}")

    # Apply Hadamard and Controlled Phase gates
    for target in range(n_qubits):

        # Apply Hadamard gate to the target qubit
        qc.h(target)

        # Apply controlled phase rotations
        for control in range(target + 1, n_qubits):

            # Calculate the rotation parameter
            k = control - target + 1

            # Calculate the controlled phase angle
            angle = 2 * np.pi / (2 ** k)

            # Apply Controlled Phase gate
            qc.cp(angle, control, target)

        # Separate circuit operations using a barrier
        qc.barrier()

    # Swap qubits to match standard bit ordering
    for i in range(n_qubits // 2):
        qc.swap(i, n_qubits - i - 1)

    # Return the QFT circuit
    return qc


# Experiment setup
n = 5

# Build the 5-qubit QFT circuit
qft_5 = build_qft_circuit(n)

# Create the inverse QFT circuit
iqft_5 = qft_5.inverse()
iqft_5.name = f"IQFT_{n}"

# Create the complete circuit with 5 qubits and 5 classical bits
full_qc = QuantumCircuit(n, n)

# Prepare the input state |5>
input_val = 5

# Convert decimal number into 5-bit binary
# Reverse the bits according to Qiskit qubit indexing
binary_repr = format(input_val, f'0{n}b')[::-1]

# Apply X gates wherever the binary bit is 1
for qubit_idx, bit in enumerate(binary_repr):
    if bit == '1':
        full_qc.x(qubit_idx)

# Add barrier after preparing input state
full_qc.barrier(label="Input State |5>")

# Append the QFT circuit
full_qc.append(qft_5, range(n))
full_qc.barrier(label="QFT")

# Append the inverse QFT circuit
full_qc.append(iqft_5, range(n))
full_qc.barrier(label="IQFT")

# Measure all qubits
full_qc.measure(range(n), range(n))

# Display QFT subcircuit
print("--- 5-Qubit QFT Subcircuit ---")
print(qft_5.draw(output="text"))

# Display complete circuit
print("\n--- Full Test Circuit ---")
print(full_qc.draw(output="text"))

# Create quantum simulator
simulator = AerSimulator()

# Compile the circuit for the simulator
compiled_circuit = transpile(full_qc, simulator)

# Execute the circuit for 1024 shots
result = simulator.run(compiled_circuit, shots=1024).result()

# Get measurement counts
counts = result.get_counts()

# Display measurement output
print("\n--- Measurement Output (1024 shots) ---")

for bitstring, count in counts.items():

    # Convert binary result into decimal
    decimal_val = int(bitstring, 2)

    # Display binary, decimal and measurement counts
    print(f"Bitstring: {bitstring} | Decimal: {decimal_val} | Counts: {count}/1024")

A. Basic Quantum Fourier Transform Questions
Q1. What is Quantum Fourier Transform (QFT)?
Ans: QFT is a quantum algorithm that transforms a quantum state from the computational basis into the Fourier basis.
Q2. What is the objective of this assignment?
Ans: To implement a 5-qubit Quantum Fourier Transform using Qiskit and verify it using the inverse QFT.
Q3. What is Fourier Transform?
Ans: Fourier Transform converts a signal from the time or original domain into the frequency domain.
Q4. What is the difference between Classical Fourier Transform and Quantum Fourier Transform?
Ans:
Classical Fourier Transform	Quantum Fourier Transform
Works on classical data	Works on quantum states
Uses classical computation	Uses quantum gates
Produces frequency representation	Produces quantum Fourier representation
Q5. Why do we use QFT in quantum computing?
Ans: QFT is used in quantum algorithms such as Shor's algorithm and quantum phase estimation.
Q6. What is the inverse QFT?
Ans: Inverse QFT reverses the operations of QFT and converts the Fourier-transformed state back into its original representation.
Q7. What is the relationship between QFT and IQFT?
Ans: They are inverse operations. Applying QFT followed by IQFT ideally restores the original quantum state.
Q8. What is the mathematical formula of QFT?
Ans:
\[ QFT|x\rangle = \frac{1}{\sqrt{N}} \sum_{y=0}^{N-1} e^{2\pi ixy/N}|y\rangle \]
Where:
•	N = Number of possible states.
•	x = Input state.
•	y = Output state.
•	i = Imaginary unit.
