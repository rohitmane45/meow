import re
from collections import defaultdict

class InvertedIndexEngine:

    def __init__(self):
        self.inverted_index = defaultdict(set)
        self.documents = {}

    def add_document(self, doc_id, text):

        self.documents[doc_id] = text

        terms = re.findall(r"\b[a-zA-Z]+\b", text.lower())

        for term in set(terms):
            self.inverted_index[term].add(doc_id)

    def display_index(self):

        print("=== INVERTED FILE (INDEX STRUCTURE) ===")

        for term in sorted(self.inverted_index.keys()):
            postings = sorted(list(self.inverted_index[term]))
            print(f"Term: {term:<15} -> Postings List: {postings}")

    def search_and(self, query):

        query_terms = re.findall(r"\b[a-zA-Z]+\b", query.lower())

        if not query_terms:
            return []

        matching_docs = set(self.inverted_index.get(query_terms[0], set()))

        for term in query_terms[1:]:
            matching_docs = matching_docs.intersection(
                self.inverted_index.get(term, set())
            )

        return sorted(list(matching_docs))

2. CODE WITH COMMENTS AND EXPLANATION
# Import required libraries
import re
from collections import defaultdict


# Create a class for Inverted Index
class InvertedIndexEngine:

    # Initialize the inverted index and documents
    def __init__(self):
        self.inverted_index = defaultdict(set)
        self.documents = {}

    # Add documents into the inverted index
    def add_document(self, doc_id, text):

        # Store document ID and its content
        self.documents[doc_id] = text

        # Convert text into lowercase and extract words
        terms = re.findall(r"\b[a-zA-Z]+\b", text.lower())

        # Add unique terms to the inverted index
        for term in set(terms):
            self.inverted_index[term].add(doc_id)

    # Display the generated inverted index
    def display_index(self):

        print("=== INVERTED FILE (INDEX STRUCTURE) ===")

        # Sort terms alphabetically
        for term in sorted(self.inverted_index.keys()):

            # Get sorted document IDs
            postings = sorted(list(self.inverted_index[term]))

            print(f"Term: {term:<15} -> Postings List: {postings}")

    # Search documents using Boolean AND operation
    def search_and(self, query):

        # Tokenize and convert query into lowercase
        query_terms = re.findall(r"\b[a-zA-Z]+\b", query.lower())

        # Return empty list if query contains no valid words
        if not query_terms:
            return []

        # Get documents containing the first query term
        matching_docs = set(
            self.inverted_index.get(query_terms[0], set())
        )

        # Find common documents containing all query terms
        for term in query_terms[1:]:

            matching_docs = matching_docs.intersection(
                self.inverted_index.get(term, set())
            )

        # Return sorted matching document IDs
        return sorted(list(matching_docs))


# Initialize the inverted index engine
engine = InvertedIndexEngine()


# Create a collection of sample documents
corpus = {
    "Doc1": "Information retrieval and natural language processing in healthcare.",
    "Doc2": "Machine learning algorithms for heart disease diagnosis.",
    "Doc3": "Processing medical records and natural language clinical data.",
    "Doc4": "Diagnosis and prediction of heart diseases using learning models."
}


# Add all documents into the inverted index
for doc_id, content in corpus.items():
    engine.add_document(doc_id, content)


# Display the generated inverted index
engine.display_index()


# Define search queries
test_queries = ["heart diagnosis", "natural language", "processing", "genomics"]

print("\n=== DOCUMENT RETRIEVAL RESULTS ===")


# Search each query and display matching documents
for q in test_queries:

    results = engine.search_and(q)

    print(f"Query: '{q}'")

    if results:

        print(f"  Matched Documents: {results}")

        # Display matching document content
        for r in results:
            print(f'    - {r}: "{corpus[r]}"')

    else:
        print("  Matched Documents: None (0 results found)")

3. SPPU PRACTICAL VIVA QUESTIONS AND ANSWERS
A. Basic Assignment Questions
Q1. What is Information Retrieval (IR)?
Ans: Information Retrieval is the process of searching and finding relevant information or documents from a large collection of documents.
Q2. What is an Inverted Index?
Ans: An inverted index is a data structure that stores each word along with the list of documents in which that word appears.
Q3. Why do we use an inverted index?
Ans: We use an inverted index to search documents quickly without scanning every document individually.
Q4. What is a posting list?
Ans: A posting list is a list of document IDs in which a particular word appears.
Example:
heart -> [Doc2, Doc4]
Q5. What is a document collection or corpus?
Ans: A corpus is a collection of documents used for indexing and searching.
Q6. What is indexing?
Ans: Indexing is the process of organizing words and their document references to make searching faster.
Q7. What is the objective of this assignment?
Ans: The objective is to implement document retrieval using an inverted index and Boolean AND searching.
Q8. What is a term in an inverted index?
Ans: A term is an individual word extracted from a document and stored in the index.

Q11. What is the use of set()?

C. Inverted Index and Searching Concepts
Q19. What is Boolean AND searching?
Ans: Boolean AND searching retrieves only those documents that contain all the search terms.
Q20. What is Boolean OR searching?
Ans: Boolean OR searching retrieves documents containing at least one of the search terms.
Q21. What is Boolean NOT searching?
Ans: Boolean NOT searching excludes documents containing a specified search term.
Q22. What is the difference between a normal index and an inverted index?
Ans:
•	Normal index: Maps documents to their words.
•	Inverted index: Maps words to their corresponding documents.
Q23. How does the program search for documents?
Ans:
1.	Takes a search query.
2.	Converts it into lowercase.
3.	Extracts individual terms.
4.	Retrieves the posting list of each term.
5.	Performs set intersection.

Q24. What happens if a searched word is not available in the index?
Ans: The program returns an empty set for that word. Therefore, Boolean AND searching returns no matching documents.
Q25. Why do we use set instead of list?
Ans: A set automatically removes duplicate document IDs and provides efficient membership and intersection operations.
Q26. Why do we use set(terms) while adding documents?
Ans: To ensure that each document ID is added only once for a particular word, even if that word appears multiple times in the document.
Q27. Can one word appear in multiple documents?
Ans: Yes. For example, the word heart appears in Doc2 and Doc4.
Q28. What is the difference between tokenization and indexing?
Ans: Tokenization extracts individual words from text, whereas indexing stores those words with their document references.
Q29. What is the time complexity of searching using an inverted index?
Ans: Searching depends mainly on the sizes of the posting lists being processed. Set intersection can be performed efficiently without scanning the entire document collection.
